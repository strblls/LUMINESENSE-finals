-- ── STEP 1: Relabel any existing head_teacher/principal admins ──
-- Before we lock the admin_role column to only allow 'admin',
-- any rows currently marked head_teacher or principal need to
-- become plain 'admin' first — otherwise step 2 fails.
UPDATE admins SET admin_role = 'admin' WHERE admin_role IN ('head_teacher', 'principal');


-- ── STEP 2: Lock admin_role to a single value ───────────────────
-- All admins are equal now. This keeps the column (in case you
-- want tiers again someday) but only 'admin' is a legal value.
ALTER TABLE admins MODIFY COLUMN admin_role ENUM('admin') NOT NULL DEFAULT 'admin';


-- ── STEP 3: Drop the duplicate FKs on admins.department_id ──────
-- You have TWO identical foreign keys doing the same thing
-- (admins_ibfk_1 and admins_ibfk_2, both department_id -> departments.id).
-- Both need to be dropped before we can drop the column itself.
ALTER TABLE admins DROP FOREIGN KEY admins_ibfk_1;
ALTER TABLE admins DROP FOREIGN KEY admins_ibfk_2;


-- ── STEP 4: Drop admins.department_id ────────────────────────────
-- This existed to scope a head_teacher to one department. Since
-- every admin can see every department now, it's dead weight.
ALTER TABLE admins DROP COLUMN department_id;


-- ── STEP 5: Add approved_by / approved_at to admins ──────────────
-- New admin signups need approval from an EXISTING admin before
-- they can log in (mirrors how faculty get approved by admins).
ALTER TABLE admins ADD COLUMN approved_by INT NULL DEFAULT NULL AFTER is_verified;
ALTER TABLE admins ADD COLUMN approved_at TIMESTAMP NULL DEFAULT NULL AFTER approved_by;
ALTER TABLE admins
    ADD CONSTRAINT fk_admin_approved_by
    FOREIGN KEY (approved_by) REFERENCES admins(id) ON DELETE SET NULL;


-- ── STEP 6: Drop departments.head_teacher_id ─────────────────────
-- This pointed at admins.id for the old head_teacher concept.
-- That concept doesn't exist anymore.
ALTER TABLE departments DROP FOREIGN KEY departments_ibfk_1;
ALTER TABLE departments DROP COLUMN head_teacher_id;

-- departments.head_faculty_id stays exactly as-is — this is the
-- one that matters: it points to faculty.id, identifying who the
-- Head Faculty is for this department. No change needed here,
-- it was already correct.


-- ── STEP 7: Drop faculty.head_teacher_id ──────────────────────────
-- Same reasoning as step 6 — the head_teacher concept is gone,
-- so this column (and its FK to admins.id) goes too.
ALTER TABLE faculty DROP FOREIGN KEY faculty_ibfk_3;
ALTER TABLE faculty DROP COLUMN head_teacher_id;


-- ── STEP 8: Drop the duplicate FK on faculty.department_id ───────
-- Same duplicate-FK issue as admins had: faculty_ibfk_2 and
-- faculty_ibfk_4 are both department_id -> departments.id.
-- Keep ONE (faculty_ibfk_2), drop the redundant one.
ALTER TABLE faculty DROP FOREIGN KEY faculty_ibfk_4;


-- ── STEP 9: Fix faculty.head_faculty_id to point at the RIGHT table ──
-- This is the important bug fix. Right now faculty_ibfk_5 points
-- head_faculty_id -> admins.id, which is wrong on two counts:
--   1. You confirmed head_faculty_id means "who is MY supervisor,"
--      i.e. another row in the FACULTY table — not an admin.
--   2. Head Faculty aren't admins anymore at all.
-- We drop the wrong FK and recreate it pointing at faculty.id
-- (a self-referencing foreign key — a faculty table pointing to
-- itself, which is normal and fine).
ALTER TABLE faculty DROP FOREIGN KEY faculty_ibfk_5;
ALTER TABLE faculty
    ADD CONSTRAINT fk_faculty_supervisor
    FOREIGN KEY (head_faculty_id) REFERENCES faculty(id) ON DELETE SET NULL;
